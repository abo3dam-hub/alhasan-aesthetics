    export const set = mutation({
      args: { key: v.string(), value: v.any() },
      handler: async (ctx, args) => {
        await requireAdmin(ctx);
        const existing = await ctx.db
          .query("siteSettings")
          .withIndex("by_key", (q) => q.eq("key", args.key))
          .first();
        if (existing) {
          await ctx.db.patch(existing._id, { value: args.value });
        } else {
          await ctx.db.insert("siteSettings", {
            key: args.key,
            value: args.value,
          });
        }
      },
    });

  What happens:
    - requireAdmin(ctx) verifies authenticated user has admin role
    - Queries siteSettings table with by_key index for key: "hero"
    - If row exists: patches the value field with new form data
    - If row doesn't exist: inserts new row with key: "hero" and value: form

  STEP 4: requireAdmin authorization check
  ─────────────────────────────────────────
  File: src/convex/admin.ts, line 7-15

    export async function requireAdmin(ctx: MutationCtx) {
      const userId = await getAuthUserId(ctx);
      if (!userId) {
        throw new Error("Not authenticated");
      }
      const user = await ctx.db.get(userId);
      if (!user || user.role !== "admin") {
        throw new Error("Unauthorized: admin access required");
      }
      return { userId, user };
    }

  What happens:
    - Gets current authenticated user ID from Convex Auth
    - Fetches user record from database
    - Checks if user.role === "admin"
    - Throws if not authenticated or not admin

     title fields (titleAr or titleEn) is an empty string, causing the
     && check to evaluate to false.

  4. UNLIKELY: Convex query caching or real-time subscription issue.
     Convex is designed for real-time reactivity and this is a core
     feature.

  5. VERY UNLIKELY: Vercel caching. The Hero component reads CMS data
     at runtime via JavaScript, not from cached HTML.
